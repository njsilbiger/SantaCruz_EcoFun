1 Overview

This Quarto document:

  • Loads and merges pH/TA/nutrients and fDOM datasets
  • Cleans data and applies outlier handling (IQR rule)
  • Summarizes by habitat and diel period
  • Visualizes diel and habitat patterns (boxplots, TA–pH, fDOM heatmap)
  • Computes correlation matrices and significance masks
  • Runs non-parametric statistics (Kruskal–Wallis, Wilcoxon, Dunn vs. control)
  • Performs PCA with scores and loadings plots
  • Lists testable hypotheses informed by the exploratory results

Reproducibility & Modularity
Kept times (which removes summer), and the control label (open ocean) are parameterized in the YAML params: so I can change inputs without editing code chunks.

# Helper to IQR-filter a numeric vector (returning NA for outliers)
iqr_filter <- function(x){
  if(!is.numeric(x)) return(x)
  q1 <- quantile(x, 0.25, na.rm = TRUE)
  q3 <- quantile(x, 0.75, na.rm = TRUE)
  iqr <- q3 - q1
  lo <- q1 - 1.5 * iqr
  hi <- q3 + 1.5 * iqr
  ifelse(x < lo | x > hi, NA_real_, x)
}

# Cleaned dataset
data_clean <- data_merged %>%

    Day_Night = factor(Day_Night, levels = c("Dawn", "Dusk"))
  ) %>%
  drop_na(Benthos, Day_Night) %>%
  mutate(across(all_of(all_numeric_vars), iqr_filter))

6 Diel & Habitat Visualizations

A helper function produces publication-ready boxplots with jitter, faceted by Season × Benthos.

#' Create diel (Dawn vs Dusk) boxplot faceted by Season × Benthos
#' @param var_to_plot Column name (string)
#' @param var_label  Y-axis label (string)
plot_diel_comparison <- function(var_to_plot, var_label){
  stopifnot(var_to_plot %in% names(data_clean))

  data_clean %>%
    filter(!is.na(.data[[var_to_plot]])) %>%
    ggplot(aes(x = Day_Night, y = .data[[var_to_plot]], fill = Day_Night)) +
    geom_boxplot(outlier.shape = NA) +
    geom_jitter(width = 0.15, alpha = 0.5, size = 2) +
    facet_wrap(Season ~ Benthos, 
              # scales = "free_y", 
               ncol = 5) +
    scale_fill_manual(values = c("Dawn" = "#FDB813", "Dusk" = "#003865")) +
    labs(title = paste("Diel Patterns in", var_label), x = "Time of Day", y = var_label) +
    theme(legend.position = "none", strip.background = element_rect(fill = "grey90"),
          axis.text.x = element_text(angle = 45, hjust = 1))
}

8 Statistical Comparisons

General testing flow: Calculate change between dawn and dusk, then perform Kruskall-Wallis test on delta(variable) and Dunn’s post-hoc to test diff from Open Ocean

# calculate Δvariable

dawn_dusk <- map_dfr(all_numeric_vars, function(v){
  data_clean %>%
    group_by(Season, Benthos, Day_Night, Quad_ID) %>%
    summarize(mean = mean(.data[[v]], na.rm = TRUE), .groups = "drop") %>%
    pivot_wider(names_from = Day_Night, values_from = mean) %>%
    mutate(
      Dusk = as.numeric(Dusk),
      Dawn = as.numeric(Dawn),
      delta_diel = Dusk - Dawn,
      variable = v)
  })

# compare with OO
diel_test <- map_dfr(all_numeric_vars, function(v){
dunn_test(delta_diel ~ Benthos, data = dawn_dusk%>% filter(Season == "Fall"), p.adjust.method = "bonferroni") %>%
    mutate(Season = "Fall")
dunn_test(delta_diel ~ Benthos, data = dawn_dusk%>% filter(Season == "Spring"), p.adjust.method = "bonferroni") %>%
    mutate(Season = "Spring")
})

  mutate(Season = as.factor(Season))

compute_sig_to_ctrl <- function(testing_df_norm, vars = all_numeric_vars_normal) {
  
  options(contrasts = c("contr.sum", "contr.poly")) # set contrasts for type III ANOVA
  
  res_tbl <- map(vars, function(v) {
    df <- testing_df_norm %>%
      filter(variable == v)
    
    fit <- lm(Value ~ Benthos * Day_Night * Season, data = df) # create lm
    model <- Anova(fit, type = "III")
    emm <- emmeans(fit, ~ Benthos | Day_Night * Season) # emmeans for tukey's hsd
    vs_ctrl <- contrast(emm, method = "trt.vs.ctrl", ref = control_label, adjust = "mvt")
    td  <- as.data.frame(vs_ctrl)
    
    comp_tbl <- td %>%
      separate(contrast, c("left","right"), sep = " - ", remove = FALSE) %>%
      mutate(
        Benthos  = if_else(right == control_label, left, right),
        variable = v,
        sig = case_when(
          is.na(p.value)  ~ NA_character_,
          p.value < 0.001 ~ "***",
          p.value < 0.01  ~ "**",
          p.value < 0.05  ~ "*",
          TRUE            ~ ""
        )
      ) 
    ctrl_tbl <- crossing(
      variable = v,
      Season   = unique(td$Season),
      Day_Night= unique(td$Day_Night),
      Benthos  = control_label
    ) %>%
      mutate(
        estimate = NA_real_, SE = NA_real_, df = NA_real_, t.ratio = NA_real_,
        p.value  = NA_real_, sig = ""
      )

    bind_rows(comp_tbl, ctrl_tbl) %>%
      distinct(variable, Season, Day_Night, Benthos, .keep_all = TRUE)
  }) %>% compact() %>% list_rbind()

  res_tbl
}

sig_to_ctrl <- compute_sig_to_ctrl(testing_df_norm) %>%
